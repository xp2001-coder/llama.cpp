cmake_minimum_required(VERSION 3.18)
project(ggml_maca_probes LANGUAGES CXX)

# This is a standalone bring-up project, not a registered ggml backend.
set(MACA_PATH "$ENV{MACA_PATH}" CACHE PATH "MACA SDK root")
option(MACA_PROBE_DEVICE "Build cu-bridge device probes" OFF)
set(MACA_CU_BRIDGE "${MACA_PATH}/tools/cu-bridge" CACHE PATH "MACA cu-bridge root")
if (NOT EXISTS "${MACA_PATH}/include/mcr/mc_runtime.h")
    message(FATAL_ERROR "MACA_PATH must contain include/mcr/mc_runtime.h")
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
get_filename_component(LLAMA_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(MACA_VENDOR_DIR "${LLAMA_SOURCE_DIR}/ggml/src/ggml-cuda/vendors")

if (MACA_PROBE_DEVICE)
    if (NOT EXISTS "${MACA_CU_BRIDGE}/include/bridge/runtime/cuda_to_maca_mcr_adaptor.h")
        message(FATAL_ERROR "MACA cu-bridge runtime headers were not found")
    endif()
    find_package(CUDAToolkit REQUIRED)
    enable_language(CUDA)
    add_executable(test-maca-bridge-runtime runtime.cpp)
    add_executable(test-maca-warp warp.cu)
    add_executable(test-maca-fp16 types.cu)
    add_executable(test-maca-bf16 types.cu)
    add_executable(test-maca-blas blas.cpp)
    target_compile_definitions(test-maca-bf16 PRIVATE MACA_PROBE_BF16)
    foreach(probe bridge-runtime warp fp16 bf16 blas)
        set(target test-maca-${probe})
        target_compile_features(${target} PRIVATE cxx_std_17)
        set_target_properties(${target} PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED YES)
        target_include_directories(${target} BEFORE PRIVATE "${MACA_CU_BRIDGE}/include" "${MACA_PATH}/include" "${MACA_VENDOR_DIR}")
        target_compile_definitions(${target} PRIVATE GGML_MACA_USE_CU_BRIDGE)
        target_link_libraries(${target} PRIVATE CUDA::cudart)
    endforeach()
    target_compile_definitions(test-maca-bridge-runtime PRIVATE MACA_PROBE_BRIDGE_RUNTIME)
    target_link_libraries(test-maca-blas PRIVATE CUDA::cublas)
else()
    find_library(MACA_RUNTIME_LIBRARY NAMES mcruntime PATHS "${MACA_PATH}/lib64" "${MACA_PATH}/lib" NO_DEFAULT_PATH REQUIRED)
    add_executable(test-maca-runtime runtime.cpp)
    target_compile_features(test-maca-runtime PRIVATE cxx_std_17)
    target_include_directories(test-maca-runtime PRIVATE "${MACA_PATH}/include" "${MACA_VENDOR_DIR}")
    target_link_libraries(test-maca-runtime PRIVATE "${MACA_RUNTIME_LIBRARY}")
endif()
